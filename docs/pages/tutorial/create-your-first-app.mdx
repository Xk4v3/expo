// App.js

import React, { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, Alert, StyleSheet } from 'react-native';

export default function App() {
  const [carType, setCarType] = useState('');
  const [price, setPrice] = useState('');
  const [address, setAddress] = useState('');
  const [paymentMethod, setPaymentMethod] = useState('');

  const handleCarType = (type) => {
    setCarType(type);
    if (type === 'სედანი') {
      setPrice('20 ლარი');
    } else if (type === 'ჯიპი') {
      setPrice('30 ლარი');
    } else {
      setPrice('');
    }
  };

  const handleOrder = () => {
    if (!carType || !address || !paymentMethod) {
      Alert.alert('გთხოვ შეავსე ყველა ველი');
    } else {
      Alert.alert('შეკვეთა მიღებულია!');
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Car Wash</Text>

      <Text style={styles.label}>აირჩიე მანქანის ტიპი:</Text>
      <View style={styles.buttonRow}>
        <TouchableOpacity
          style={[styles.button, carType === 'სედანი' && styles.selected]}
          onPress={() => handleCarType('სედანი')}>
          <Text style={styles.buttonText}>სედანი</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.button, carType === 'ჯიპი' && styles.selected]}
          onPress={() => handleCarType('ჯიპი')}>
          <Text style={styles.buttonText}>ჯიპი</Text>
        </TouchableOpacity>
      </View>

      <Text style={styles.label}>ფასი: {price}</Text>

      <Text style={styles.label}>შეიყვანე მისამართი:</Text>
      <TextInput
        style={styles.input}
        placeholder="მაგ: რუსთავი, მესხეთის 4"
        value={address}
        onChangeText={setAddress}
      />

      <Text style={styles.label}>გადახდის მეთოდი:</Text>
      <View style={styles.buttonRow}>
        <TouchableOpacity
          style={[styles.button, paymentMethod === 'ქეში' && styles.selected]}
          onPress={() => setPaymentMethod('ქეში')}>
          <Text style={styles.buttonText}>ქეში</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.button, paymentMethod === 'ბარათი' && styles.disabled]}
          disabled={true}>
          <Text style={styles.buttonText}>ბარათი (დროებით არ მუშაობს)</Text>
        </TouchableOpacity>
      </View>

      <TouchableOpacity style={styles.orderButton} onPress={handleOrder}>
        <Text style={styles.orderText}>შეკვეთა</Text>
      </TouchableOpacity>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, justifyContent: 'center', backgroundColor: '#f0f0f0' },
  title: { fontSize: 28, fontWeight: 'bold', marginBottom: 30, textAlign: 'center' },
  label: { fontSize: 16, marginBottom: 5 },
  input: { backgroundColor: '#fff', padding: 10, marginBottom: 15, borderRadius: 10 },
  buttonRow: { flexDirection: 'row', justifyContent: 'space-around', marginBottom: 15 },
  button: { padding: 10, borderRadius: 10, backgroundColor: '#ddd' },
  selected: { backgroundColor: '#4caf50' },
  disabled: { backgroundColor: '#aaa' },
  buttonText: { color: '#000' },
  orderButton: { marginTop: 20, backgroundColor: '#2196f3', padding: 15, borderRadius: 10 },
  orderText: { textAlign: 'center', color: '#fff', fontSize: 16, fontWeight: 'bold' },
});
